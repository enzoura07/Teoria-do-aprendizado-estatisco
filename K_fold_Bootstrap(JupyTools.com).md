set.seed(123)

k <- 5

indices <- sample(
  1:nrow(dados_modelo)

folds <- split(
  indices,
  cut(
    seq_along(indices),
    breaks = k,
    labels = FALSE
  )

sapply(folds, length)

```R
install.packages("pROC")
library(pROC)

# BLOCO 7B - MODELO 1

resultados_modelo1 <- data.frame()

for(i in 1:k){

  treino <- dados_modelo[-folds[[i]], ]
  teste <- dados_modelo[folds[[i]], ]

  modelo1 <- glm(
    causa_alvo ~ IDADE + SEXO,
    data = treino,
    family = binomial
  )

  prob <- predict(
    modelo1,
    newdata = teste,
    type = "response"
  )

  resultado <- data.frame(
    Fold = i,
    AUC = as.numeric(
      roc(
        teste$causa_alvo,
        prob,
        quiet = TRUE
      )$auc
    )
  )

  resultados_modelo1 <- rbind(
    resultados_modelo1,
    resultado
  )

  cat("Fold", i, "concluído\n")
}

resultados_modelo1

# BLOCO 9A - FUNÇÃO BOOTSTRAP


funcao_bootstrap <- function(data, indices){

  amostra <- data[indices, ]

  modelo <- glm(
    causa_alvo ~ IDADE + SEXO + RAÇA + ESCOLARIDADE + LOCAL_MORTE,
    data = amostra,
    family = binomial
  )

  return(
    coef(modelo)
  )
}
```

# BLOCO 10 - RESULTADO DO BOOTSTRAP


coef_simulations <- bootstrap_resultado$t[, selected_coef_index]

erro_padrao <- sd(
  coef_simulations,
  na.rm = TRUE
)

IC <- quantile(
  coef_simulations,
  probs = c(0.025, 0.975),
  na.rm = TRUE
)

cat(
  "Erro padrão Bootstrap para", selected_coefficient_name, ":",
  erro_padrao,
  "\n"
)

cat(
  "IC 95% inferior para", selected_coefficient_name, ":",
  IC[1],
  "\n"
)

cat(
  "IC 95% superior para", selected_coefficient_name, ":",
  IC[2],
  "\n"
)
